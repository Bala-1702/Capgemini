#include <stdio.h>
#include<stdlib.h>
struct node{
    int data;
    struct node * next;
};
void create(struct node** head,int data)
{
    struct node * newNode=(struct node*)malloc(sizeof(struct node));
    newNode->data=data;
    newNode->next=NULL;
    if(*head==NULL)
    {
        *head=newNode;
        return;
    }
    struct node * temp = *head;
    while(temp->next!=NULL)
    {
        temp=temp->next;
    }
    temp->next=newNode;
}
void printList(struct node** head)
{
    struct node * temp = *head;
    while(temp!=NULL)
    {
        printf("%d->",temp->data);
        temp=temp->next;
    }
    printf("NULL");
}
int main()
{
    struct node* head=NULL;
    create(&head,1);
    create(&head,2);
    create(&head,3);
    create(&head,4);
    create(&head,5);
    printList(&head);
    return 0;
}

int dir1[4]={1,0,-1,0};
int dir2[4]={0,1,0,-1};
int visit[9][9];
int solve(int i,int j,int mat[9][9],int row,int col,int visit[9][9])
	{
	    if(mat[i][j]==9)
	        return  1;
	    
	    
	    int result=0;
	    visit[i][j]=1;
	    for(int k=0;k<4;k++)
	    {
	        int x=i+dir1[k];
	        int y=j+dir2[k];
	        if(x>-1 && y>-1 && x<row && y<col && mat[x][y]!=0 && visit[x][y]==0)
	        {
	            result=result||solve(x,y,mat,row,col,visit);
	        }
	    }
	    visit[i][j]=0;
	    return result;
	    
	}
